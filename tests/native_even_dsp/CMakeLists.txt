cmake_minimum_required(VERSION 3.22)
project(native_even_dsp_tests LANGUAGES CXX)
add_executable(native_even_dsp_test ../native_even_dsp_test.cpp)
target_compile_features(native_even_dsp_test PRIVATE cxx_std_17)
target_include_directories(native_even_dsp_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../..")
if(MSVC)
    target_compile_options(native_even_dsp_test PRIVATE /O2 /W4 /EHsc)
else()
    target_compile_options(native_even_dsp_test PRIVATE -O2 -Wall -Wextra)
endif()
