cmake_minimum_required(VERSION 3.22)
project(reuse_live_dsp LANGUAGES CXX)
add_library(reuse_live_dsp SHARED resampler_jni.cpp)
target_compile_features(reuse_live_dsp PRIVATE cxx_std_17)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(reuse_live_dsp PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(MSVC)
  target_compile_options(reuse_live_dsp PRIVATE /O2 /W4 /EHsc)
else()
  target_compile_options(reuse_live_dsp PRIVATE -O3 -Wall -Wextra -fvisibility=hidden)
endif()
if(ANDROID)
  target_link_options(reuse_live_dsp PRIVATE "-Wl,-z,max-page-size=16384")
endif()
