cmake_minimum_required(VERSION 3.22)
project(reuse_ort_ops LANGUAGES CXX)

option(REUSE_EXACT_EXP "Use scalar std::exp instead of the SIMD exponential" OFF)
option(REUSE_NEON_FMA "Experimental explicit ARM NEON FMA arithmetic; requires fresh parity/performance validation" OFF)
option(REUSE_SCAN_N16_REGISTERS "Experimental rank16 frequency-scan register state instead of scratch traffic" OFF)
set(REUSE_SCAN_CHANNEL_GRAIN "16" CACHE STRING "Frequency Scan channels per parallel callback (8,16,32,64); arithmetic unchanged")
set_property(CACHE REUSE_SCAN_CHANNEL_GRAIN PROPERTY STRINGS 8 16 32 64)
if(NOT REUSE_SCAN_CHANNEL_GRAIN MATCHES "^(8|16|32|64)$")
  message(FATAL_ERROR "REUSE_SCAN_CHANNEL_GRAIN must be 8,16,32,64")
endif()
option(REUSE_BUILD_KERNEL_BENCHMARK "Build isolated native kernel correctness/timing executable" OFF)
add_library(reuse_ort_ops SHARED selective_scan.cpp)
target_compile_features(reuse_ort_ops PRIVATE cxx_std_17)
target_include_directories(reuse_ort_ops PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(reuse_ort_ops PRIVATE ORT_API_MANUAL_INIT)
if(REUSE_EXACT_EXP)
  target_compile_definitions(reuse_ort_ops PRIVATE REUSE_EXACT_EXP=1)
endif()
if(REUSE_NEON_FMA)
  target_compile_definitions(reuse_ort_ops PRIVATE REUSE_NEON_FMA=1)
endif()
if(REUSE_SCAN_N16_REGISTERS)
  target_compile_definitions(reuse_ort_ops PRIVATE REUSE_SCAN_N16_REGISTERS=1)
endif()
if(NOT REUSE_SCAN_CHANNEL_GRAIN STREQUAL "16")
  target_compile_definitions(reuse_ort_ops PRIVATE REUSE_SCAN_CHANNEL_GRAIN=${REUSE_SCAN_CHANNEL_GRAIN})
endif()
if(MSVC)
  target_compile_options(reuse_ort_ops PRIVATE /O2 /W4 /EHsc)
else()
  target_compile_options(reuse_ort_ops PRIVATE -O3 -Wall -Wextra -fvisibility=hidden)
endif()
if(ANDROID)
  target_sources(reuse_ort_ops PRIVATE runtime_build_info_jni.cpp)
  target_link_libraries(reuse_ort_ops PRIVATE ${CMAKE_DL_LIBS})
  target_link_options(reuse_ort_ops PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(REUSE_BUILD_KERNEL_BENCHMARK)
  add_executable(reuse_kernel_benchmark ../tools/native_kernel_benchmark.cpp)
  target_compile_features(reuse_kernel_benchmark PRIVATE cxx_std_17)
  target_include_directories(reuse_kernel_benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_definitions(reuse_kernel_benchmark PRIVATE ORT_API_MANUAL_INIT)
  if(NOT REUSE_SCAN_CHANNEL_GRAIN STREQUAL "16")
    target_compile_definitions(reuse_kernel_benchmark PRIVATE REUSE_SCAN_CHANNEL_GRAIN=${REUSE_SCAN_CHANNEL_GRAIN})
  endif()
  foreach(experiment REUSE_EXACT_EXP REUSE_NEON_FMA REUSE_SCAN_N16_REGISTERS)
    if(${experiment})
      target_compile_definitions(reuse_kernel_benchmark PRIVATE ${experiment}=1)
    endif()
  endforeach()
  if(MSVC)
    target_compile_options(reuse_kernel_benchmark PRIVATE /O2 /W4 /EHsc)
  else()
    target_compile_options(reuse_kernel_benchmark PRIVATE -O3 -Wall -Wextra)
  endif()
endif()
# The host runtime passes the C API to RegisterCustomOps. There is no link-time
# dependency on an ORT DLL/.so, so this target works for Windows and Android.
